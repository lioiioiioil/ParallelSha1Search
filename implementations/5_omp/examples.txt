1. OpenMP
Все команды выполняются из папки этого задания.
Ввод: правильный SHA1 в нижнем регистре. Ответ примера: 000000a.

MAC: нужна libomp (brew install libomp)
c++ -O2 -std=c++17 -Xpreprocessor -fopenmp -I"$(brew --prefix libomp)/include" main.cpp -L"$(brew --prefix libomp)/lib" -Wl,-rpath,"$(brew --prefix libomp)/lib" -lomp -o omp
echo 28a2f4f4403abf7b4e9bea6683a82ad529aca211 | ./omp

LINUX: компилятор с OpenMP
g++ -O2 -std=c++17 -fopenmp main.cpp -o omp
echo 28a2f4f4403abf7b4e9bea6683a82ad529aca211 | ./omp

WINDOWS: Developer Command Prompt
cl /nologo /O2 /std:c++17 /EHsc /utf-8 /openmp main.cpp /Fe:omp.exe
echo 28a2f4f4403abf7b4e9bea6683a82ad529aca211| omp.exe
