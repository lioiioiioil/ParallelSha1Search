cmake_minimum_required(VERSION 3.18)
project(ParallelSha1Search LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
    add_compile_options(/utf-8)
endif()
find_package(Threads REQUIRED)
add_executable(serial implementations/5_brute/main.cpp)
add_executable(std_thread implementations/std_thread/main.cpp)
target_link_libraries(std_thread PRIVATE Threads::Threads)
if(UNIX)
    add_executable(pthread_search implementations/pthread_create/main.cpp)
    target_link_libraries(pthread_search PRIVATE Threads::Threads)
    add_executable(fork_search implementations/fork/main.cpp)
endif()
if(WIN32)
    add_executable(create_thread implementations/CreateThread/main.cpp)
    add_executable(create_process implementations/CreateProcess/main.cpp)
endif()
option(ENABLE_OPENMP "Build OpenMP implementation" OFF)
option(ENABLE_MPI "Build MPI implementations" OFF)
option(ENABLE_CUDA "Build CUDA implementation" OFF)
if(ENABLE_OPENMP)
    find_package(OpenMP REQUIRED)
    add_executable(omp_search implementations/5_omp/main.cpp)
    target_link_libraries(omp_search PRIVATE OpenMP::OpenMP_CXX)
endif()
if(ENABLE_MPI)
    find_package(MPI REQUIRED COMPONENTS CXX)
    add_executable(mpi_2 implementations/MPI_2/main.cpp)
    add_executable(mpi_collective implementations/MPI_collective/main.cpp)
    target_link_libraries(mpi_2 PRIVATE MPI::MPI_CXX)
    target_link_libraries(mpi_collective PRIVATE MPI::MPI_CXX)
endif()
if(ENABLE_CUDA)
    enable_language(CUDA)
    add_executable(cuda_search implementations/cuda/main.cu)
    set_target_properties(cuda_search PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
endif()
